script void detAttempt(WsfPlatform aPlatform, WsfSensor aSensor, WsfPlatform aTarget, WsfSensorInteraction aResult)
   writeln("SNR ", aTarget.Name(), " ", aResult.SignalToNoise());
end_script

observer
   enable SENSOR_DETECTION_ATTEMPT detAttempt
end_observer

processor jam_mgr WSF_TASK_PROCESSOR
   script bool AddSpot()
      bool started = false;
      for (int i = 0; i < PLATFORM.WeaponCount(); i = i + 1)
      {
         WsfWeapon j = PLATFORM.WeaponEntry(i);
         if (j.CanJam(3300000000.000000))
         {
            started = StartJamming(TRACK, "jam1", j, 3300000000.000000, 2000000.000000, "noise_jamming");
         }
      }
      return started;
   end_script

   evaluation_interval DETECTED 2.0 sec
   state DETECTED
      next_state JAMMING
         return AddSpot();
      end_next_state
   end_state
   evaluation_interval JAMMING 2.0 sec
   state JAMMING
      next_state DETECTED
         return false;
      end_next_state
   end_state
end_processor
